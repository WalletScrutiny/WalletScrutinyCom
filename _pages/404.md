<script>
document.getElementById('notFoundSearch')?.addEventListener('click', (event) => {
  event.preventDefault();

  const activateSearch = () => {
    const walletSearch = document.querySelector('.wallet-search');
    const searchbar = document.querySelector('.searchbar');
    if (!searchbar) return;

    if (window.innerWidth <= 700 && walletSearch && !walletSearch.classList.contains('mobile-active')) {
      walletSearch.classList.add('mobile-active');
      document.querySelector('.mobile-search-shortcut')?.classList.add('active');
    }

    searchbar.focus({ preventScroll: true });
  };

  if (window.scrollY === 0) {
    activateSearch();
    return;
  }

  window.scrollTo({ top: 0, left: 0, behavior: 'smooth' });

  if ('onscrollend' in window) {
    window.addEventListener('scrollend', activateSearch, { once: true });
  } else {
    window.setTimeout(activateSearch, 500);
  }
});
</script>
